-- Creating Movie Table
Create table Movie ( movie_id int auto_increment primary key,
Title varchar(100), Release_year int);

-- Creating Media Table
create table Media (media_id int auto_increment primary key,
movie_id int,
Media_type Enum('Video', 'Image') NOT null, 
url varchar (300) not null,
foreign key (movie_id) references Movie(movie_id)
);

-- Join Query
select m.Title, m.Release_year, med.media_type,a.artist_name, r.role_name from Movie m 
inner join Media med on med.movie_id = m.movie_id
inner join Artist_roles ar on ar.movie_id = m.movie_id
inner join Artist a on a.artist_id = ar.artist_id
inner join Roles r on r.role_id = ar.role_id;

-- Movie should have multiple media(video or image)
select m.Title, m.Release_year,md.Media_type, md.url from Movie m 
join media md on md.movie_id = m.movie_id;

-- Movie can belongs to Multiple Genre
select m.title,g.name from Movie m
join movie_genre mg on m.movie_id = mg.movie_id
join Genre g on g.genre_id = mg.genre_id;

-- Movie can have multiple reviews and Review can belongs to user
select m.title, m.Release_year,u.user_name, r.rating, r.comment from movie m
inner join review r on r.movie_id
inner join user u on u.user_id = r.user_id;

-- Artist can have multiple skills
select a.Artist_name,s.skill_name from Artist_skills artskil
join Artist a on a.artist_id = artskil.artist_id
join skill s on s.skill_id = artskil.skill_id;

-- Artist can perform multiple role in single film
select m.title,a.artist_name,r.Role_name from Artist_roles ar
join movie m on m.movie_id = ar.movie_id
join roles r on r.role_id = ar.role_id
join artist a on a.artist_id = ar.artist_id;
